# Stage2: 오른쪽으로 오래 걷기. 카메라가 기존과 같은 경계에서 멈추는지
# camera.x/y = 화면 가운데가 보는 월드 좌표. 해상도 1600x900이라 화면 왼쪽 끝은 camera.x - 800.
# 기준 동작(2026-10-23, Camera::SetBounds 도입 전):
#   - 왼쪽: player.x < 800이면 camera.x = 800 (화면 왼쪽 끝 0)
#   - 오른쪽: 816프레임쯤부터 camera.x = 2550 (화면 오른쪽 끝 3350)에서 멈춤
#   - 바닥에 서 있으면 camera.y = 1501 (player.y 1726 - 225)
#   - 플레이어는 바닥(x 0~3400)을 지나 약 4000까지 걸어간다
scene Stage2
frames 1300

hold RIGHT 50 1250

expect 50   camera.x == 800
expect 50   camera.y == 1501
expect 500  camera.x >= 1490        # 가운데 구간은 플레이어를 따라감
expect 900  camera.x == 2550
expect 1250 player.x >= 3900
expect 1250 camera.x == 2550
expect 1299 camera.y == 1501
always      camera.x >= 800
always      camera.x <= 2550
always      camera.y <= 1505        # 착지 순간 player.y가 1730까지 파고들 때 1505
shot 1299
